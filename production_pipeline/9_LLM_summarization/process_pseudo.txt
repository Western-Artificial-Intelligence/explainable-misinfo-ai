# 9_LLM_summarization/process_pseudo.txt
# Goal: produce a concise, user-facing summary using a small LLM, grounded in:
#   - RoBERTa verdict + confidence
#   - evidence_topk (citations)
#   - SHAP top tokens (explainability)
#
# Assumption: ai_model(system_text: string, input_text: string) -> string exists.
# If your runtime supports explicit sampling params, set them per LLM_PARAMS below.
# Otherwise, include them in SYS_INSTRUCTION as a hard constraint and log a warning.

INPUT:
  {
    request_id,
    claim_id,
    user_claim,
    normalized_claim,
    roberta: { label: { class_name, probs }, confidence },
    evidence_topk: { items: [ {text, relevance_score, doc:{doc_id,source,title,url,published_at?}} ... ] },
    shap_explainability: { tokens: [...], top_tokens: {positive:[idx], negative:[idx]} },
    meta
  }

CONSTANTS (tunable):
  MAX_BULLETS = 4
  MAX_EVIDENCE_SNIPPETS = 4           # how many evidence chunks we show the LLM
  MAX_SNIP_CHARS = 400                # truncate evidence text for prompt size
  TOPK_TOKENS = 5                     # tokens to show from SHAP

  # LLM stability params (best practice)
  LLM_PARAMS = { temperature: 0.0, top_p: 1.0 }

  # Requested JSON payload from the LLM (validated strictly).
  # NOTE: used_doc_ids MUST be subset of evidence_topk doc_ids.
  LLM_JSON_SCHEMA =
    {
      "type":"object",
      "additionalProperties":false,
      "required":["short","bullets","used_doc_ids","evidence_quality"],
      "properties":{
        "short":{"type":"string","minLength":1},
        "bullets":{"type":"array","items":{"type":"string","minLength":1},"minItems":0,"maxItems":6},
        "used_doc_ids":{"type":"array","items":{"type":"string","minLength":1},"minItems":0},
        "evidence_quality":{"type":"string","enum":["strong","medium","weak"]}
      }
    }

  SYS_INSTRUCTION =
    "You are a concise fact-check assistant. " +
    "Use ONLY the provided evidence snippets and model output. " +
    "Do not invent new facts. " +
    "If evidence is weak or missing, say so clearly. " +
    "Output STRICT JSON matching the requested format. " +
    "Sampling params must be temperature=0 and top_p=1."

  # Retry policy (best practice)
  MAX_JSON_REPAIR_RETRIES = 1

  # Repair prompt (system text) used only when JSON parse/schema validation fails
  SYS_REPAIR_INSTRUCTION =
    "You must output ONLY valid JSON (no markdown, no commentary). " +
    "Fix the user's provided JSON so it conforms EXACTLY to the schema. " +
    "Do not add facts. Do not add keys. Do not include trailing text."

HELPERS (assumed):
  take_unique_by_doc(items) -> list                      # keep best chunk per doc_id (highest relevance)
  truncate(s, max_chars) -> string
  safe_json_parse(s) -> object|null                      # tolerant JSON parse
  validate_json_schema(obj, schema) -> bool              # strict JSON Schema validation
  json_dump(x) -> string
  unique_preserve_order(list) -> list
  clamp_int(x, lo, hi) -> int
  round(x, ndigits) -> number

PROCESS:
1) Determine verdict + confidence from RoBERTa (no modification here)
   verdict = roberta.label.class_name
   confidence = roberta.confidence

2) Prepare citations + evidence snippets (grounding)
   chunks = evidence_topk.items
   # deterministic: relevance_score DESC, then rank ASC
   chunks_sorted = sort(chunks by relevance_score DESC, then rank ASC)
   chunks_unique = take_unique_by_doc(chunks_sorted)
   chosen = chunks_unique[0:MAX_EVIDENCE_SNIPPETS]

   evidence_snips = []
   citations = []         # unique sources that correspond to chosen snippets
   seen_doc = set()

   FOR each c in chosen:
       snip = truncate(c.text, MAX_SNIP_CHARS)
       evidence_snips.append({
         doc_id: c.doc.doc_id,
         source: c.doc.source,
         title: c.doc.title,
         url: c.doc.url,
         published_at: c.doc.published_at?,
         relevance_score: c.relevance_score,
         snippet: snip
       })

       IF c.doc.doc_id not in seen_doc:
           seen_doc.add(c.doc.doc_id)
           citations.append({
             doc_id: c.doc.doc_id,
             source: c.doc.source,
             title: c.doc.title,
             url: c.doc.url,
             published_at: c.doc.published_at?
           })

3) Prepare SHAP token highlights (human-readable)
   pos_idx = shap_explainability.top_tokens.positive
   neg_idx = shap_explainability.top_tokens.negative
   tokens = shap_explainability.tokens

   top_pos_tokens = []
   top_neg_tokens = []

   FOR i in pos_idx[0:TOPK_TOKENS]:
       IF i < len(tokens): top_pos_tokens.append(tokens[i].token)
   FOR i in neg_idx[0:TOPK_TOKENS]:
       IF i < len(tokens): top_neg_tokens.append(tokens[i].token)

4) Build LLM payload (single string)
   payload =
     "CLAIM:\n" + user_claim + "\n\n" +
     "MODEL_VERDICT:\n" +
     "- verdict: " + verdict + "\n" +
     "- confidence: " + str(confidence) + "\n\n" +
     "EVIDENCE_SNIPPETS:\n" + json_dump(evidence_snips) + "\n\n" +
     "SHAP_TOP_TOKENS:\n" +
     "- positive_tokens: " + json_dump(top_pos_tokens) + "\n" +
     "- negative_tokens: " + json_dump(top_neg_tokens) + "\n\n" +
     "RESPONSE_SCHEMA:\n" + json_dump(LLM_JSON_SCHEMA) + "\n\n" +
     "RESPONSE_FORMAT:\n" +
     "{ \"short\": string, \"bullets\": string[], \"used_doc_ids\": string[], \"evidence_quality\": \"strong|medium|weak\" }"

5) Call LLM (stable params)
   # If runtime supports explicit params:
   #   llm_raw = ai_model(SYS_INSTRUCTION, payload, LLM_PARAMS)
   # Otherwise:
   llm_raw = ai_model(SYS_INSTRUCTION, payload)

6) Parse + STRICT schema validation, with at most 1 repair retry
   attempt = 0
   llm_obj = null

   WHILE attempt <= MAX_JSON_REPAIR_RETRIES:
       llm_obj = safe_json_parse(llm_raw)

       IF llm_obj != null AND validate_json_schema(llm_obj, LLM_JSON_SCHEMA) == true:
           break

       IF attempt == MAX_JSON_REPAIR_RETRIES:
           llm_obj = null
           break

       # Repair attempt: force JSON-only output conforming to schema
       repair_payload =
         "INVALID_OUTPUT:\n" + llm_raw + "\n\n" +
         "SCHEMA:\n" + json_dump(LLM_JSON_SCHEMA)

       llm_raw = ai_model(SYS_REPAIR_INSTRUCTION, repair_payload)
       attempt = attempt + 1

7) Fallback if still invalid
   IF llm_obj is null:
       llm_obj = {
         short: verdict + " (confidence " + str(round(confidence,2)) + "). Evidence limited; review sources.",
         bullets: [],
         used_doc_ids: [c.doc_id for c in citations],
         evidence_quality: ("weak" if len(citations)==0 else "medium")
       }

8) Enforce bounds + grounding restrictions
   # bullets bound (hard cap)
   bullets = llm_obj.bullets[0:MAX_BULLETS]

   # restrict doc_ids to ones we actually have
   allowed_doc_ids = set([c.doc_id for c in citations])
   used_doc_ids = [d for d in llm_obj.used_doc_ids if d in allowed_doc_ids]
   used_doc_ids = unique_preserve_order(used_doc_ids)

   # If model returned none, default to all citations (transparent)
   IF len(used_doc_ids) == 0:
       used_doc_ids = [c.doc_id for c in citations]

9) Build final summary block (matches output_schema.summary)
   summary = {
     verdict: verdict,
     confidence: confidence,
     short: llm_obj.short,
     bullets: bullets,
     citations: [c for c in citations if c.doc_id in set(used_doc_ids)],
     explainability: {
       top_positive_tokens: top_pos_tokens,
       top_negative_tokens: top_neg_tokens
     }
   }

10) Emit output (passthrough + summary)
   OUTPUT = INPUT plus "summary" field
   RETURN OUTPUT

DETERMINISM / TIE-BREAKING (PROD)

Definitions:
  EPS = 1e-9

Rules:
  - Evidence selection sorting is deterministic:
      relevance_score DESC, then rank ASC
  - Citations preserve chosen evidence order, filtered by used_doc_ids (stable)
