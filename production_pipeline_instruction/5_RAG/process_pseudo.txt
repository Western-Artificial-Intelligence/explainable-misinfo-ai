# 5_RAG_retrieval/process_pseudo.txt
# Goal: Use Google Search API to build a candidate pool of paragraph-based chunks (~M).
# NOTE: No MMR here. No final top-K here. This step only returns rag_candidates.

# Assumptions (implementation details ignored):
#   is_http_or_https(url: string) -> bool
#   resolve_host(url: string) -> { hostname, ips: list<string>, port: int }
#   is_private_ip(ip: string) -> bool            # RFC1918, loopback, link-local, etc.
#   has_disallowed_port(port: int) -> bool       # allowlist ports, e.g. 80/443 only
#
#   google_search_api(query: string, num_results: int) -> list<SearchResult>
#     SearchResult = { url, title, snippet, source?, published_at? }
#
#   cache_get(key: string) -> optional<bytes>
#   cache_set(key: string, value: bytes, ttl_seconds: int) -> void
#
#   rate_limit_acquire(domain: string, rps: float) -> void
#     blocks (or sleeps) to enforce max requests/sec per domain when rps > 0
#
#   fetch_url(
#       url: string,
#       connect_timeout_ms: int,
#       read_timeout_ms: int,
#       max_response_bytes: int,
#       headers: map<string,string>
#   ) -> { status_code, headers, body_bytes }
#
#   is_allowed_by_robots(url: string, user_agent: string) -> bool
#   is_paywalled(html_or_text: string) -> bool          # heuristic
#   is_js_heavy(html: string) -> bool                   # heuristic (e.g., empty body, app shell)
#   reader_fallback(url: string) -> { status_code, text }  # optional, for js_heavy_policy="fallback_reader"
#
#   decode_html(body_bytes: bytes, content_type: string) -> string
#   extract_main_text_with_paragraphs(html: string) -> list<string>
#   extract_main_text_plain(html: string) -> string     # optional fallback
#
#   normalize_text(s: string) -> string
#   merge_short_paragraphs(paras: list<string>, min_chars: int) -> list<string>
#   split_long_paragraphs(paras: list<string>, max_chars: int) -> list<string>   # optional
#   window_chunk(text: string, max_chars: int, overlap_chars: int) -> list<string>
#
#   score_chunk(claim: string, chunk: string) -> float
#   domain(url: string) -> string
#   hash_text(s: string) -> string
#   make_doc_id(url: string) -> string
#   make_chunk_id(claim_id: string, doc_id: string, p_start: int, p_end: int) -> string

INPUT:
  {
    request_id,
    claim_id,
    user_claim,
    normalized_claim,
    roberta,
    routing: {
      rag: {
        candidate_pool_size_m (M),
        allow_web_search,

        # safety / robustness knobs (all optional; defaults below)
        connect_timeout_ms?,
        read_timeout_ms?,
        max_response_bytes?,
        allowed_content_types?,
        allow_pdf?,
        respect_robots?,
        paywall_policy?,     # "skip" | "extract_snippet" | "allow"
        js_heavy_policy?,    # "skip" | "fallback_reader" | "allow"
        per_domain_rps?,
        cache_enabled?,
        cache_ttl_s?
      }
    },
    query_plan: { queries: [{q, variant}...] },
    meta
  }

CONSTANTS (tunable defaults if routing.rag.* not provided):
  PER_QUERY_RESULTS = 10
  MAX_URLS_TOTAL = 30

  # extraction tuning
  PARA_MIN_CHARS = 200
  PARA_MAX_CHARS = 1200
  FALLBACK_WINDOW_MAX = 900
  FALLBACK_WINDOW_OVERLAP = 150
  MIN_PARAS_FOR_SUCCESS = 2
  MIN_TEXT_CHARS_FALLBACK = 500

  # fetch safety defaults
  DEFAULT_CONNECT_TIMEOUT_MS = 2000
  DEFAULT_READ_TIMEOUT_MS = 6000
  DEFAULT_MAX_RESPONSE_BYTES = 2_000_000         # ~2 MB cap
  DEFAULT_ALLOWED_CT = ["text/html", "application/xhtml+xml"]
  DEFAULT_ALLOW_PDF = false
  DEFAULT_RESPECT_ROBOTS = true
  DEFAULT_PAYWALL_POLICY = "skip"
  DEFAULT_JS_HEAVY_POLICY = "skip"
  DEFAULT_PER_DOMAIN_RPS = 0                     # 0 => no explicit rate limit
  DEFAULT_CACHE_ENABLED = true
  DEFAULT_CACHE_TTL_S = 3600
  USER_AGENT = "TruthLensBot/1.0 (+mailto:ops@truthlens.ai)"

PROCESS:
1) Check retrieval allowed
   M = routing.rag.candidate_pool_size_m
   IF routing.rag.allow_web_search == false:
       rag_candidates = { pool_size_m: M, sources: ["web"], items: [] }
       OUTPUT = INPUT + rag_candidates
       RETURN OUTPUT

2) Resolve safety config (apply defaults)
   connect_timeout_ms = routing.rag.connect_timeout_ms OR DEFAULT_CONNECT_TIMEOUT_MS
   read_timeout_ms    = routing.rag.read_timeout_ms    OR DEFAULT_READ_TIMEOUT_MS
   max_bytes          = routing.rag.max_response_bytes OR DEFAULT_MAX_RESPONSE_BYTES

   allowed_cts        = routing.rag.allowed_content_types OR DEFAULT_ALLOWED_CT
   allow_pdf          = routing.rag.allow_pdf OR DEFAULT_ALLOW_PDF
   respect_robots     = routing.rag.respect_robots OR DEFAULT_RESPECT_ROBOTS

   paywall_policy     = routing.rag.paywall_policy OR DEFAULT_PAYWALL_POLICY
   js_heavy_policy    = routing.rag.js_heavy_policy OR DEFAULT_JS_HEAVY_POLICY

   per_domain_rps     = routing.rag.per_domain_rps OR DEFAULT_PER_DOMAIN_RPS
   cache_enabled      = routing.rag.cache_enabled OR DEFAULT_CACHE_ENABLED
   cache_ttl_s        = routing.rag.cache_ttl_s OR DEFAULT_CACHE_TTL_S

3) Collect search hits (URLs) using Google Search API
   queries = query_plan.queries
   url_hits = []                    # {url,title,snippet,source?,published_at?,query_used}
   seen_urls = set()

   FOR each qobj in queries:
       results = google_search_api(qobj.q, PER_QUERY_RESULTS)
       FOR each r in results:
           IF len(url_hits) >= MAX_URLS_TOTAL: break
           IF r.url in seen_urls: continue
           seen_urls.add(r.url)
           url_hits.append({
             url: r.url,
             title: r.title,
             snippet: r.snippet,
             source: r.source?,
             published_at: r.published_at?,
             query_used: qobj.q
           })

4) Fetch pages with safety checks, extract paragraphs, create candidate chunks
   candidate_items = []
   seen_chunk_hashes = set()

   FOR each hit in url_hits:
       url = hit.url
       dom = domain(url)
       
       # URL safety (SSRF hardening)
       # - Only allow http(s)
       # - Block localhost / private IP ranges (including via DNS)
       # - Optionally block non-standard ports
       IF is_http_or_https(url) == false:
           continue
       resolved = resolve_host(url)   # may perform DNS resolution
       IF has_disallowed_port(resolved.port):
           continue
       IF resolved.hostname in ["localhost", "127.0.0.1", "::1"]:
           continue
       IF any(is_private_ip(ip) for ip in resolved.ips):
           continue

       # Robots policy
       IF respect_robots == true:
           IF is_allowed_by_robots(url, USER_AGENT) == false:
               continue

       # Per-domain rate limit
       IF per_domain_rps > 0:
           rate_limit_acquire(dom, per_domain_rps)

       # Caching (raw response bytes) — key includes url + max_bytes to avoid mismatch
       cache_key = "fetch:" + url + ":max=" + str(max_bytes)
       cached = (cache_get(cache_key) if cache_enabled else null)

       IF cached exists:
           resp = { status_code: 200, headers: {"content-type": "text/html"}, body_bytes: cached }
       ELSE:
           resp = fetch_url(
             url,
             connect_timeout_ms,
             read_timeout_ms,
             max_bytes,
             headers={"User-Agent": USER_AGENT, "Accept": join(allowed_cts, ",")}
           )
           IF resp.status_code not in 200..299:
               continue

           # Cache only successful, non-empty bodies
           IF cache_enabled and len(resp.body_bytes) > 0:
               cache_set(cache_key, resp.body_bytes, cache_ttl_s)

       # Content-type checks
       ct = lower(resp.headers.get("content-type", ""))

       # PDF handling
       IF ("application/pdf" in ct) OR url endswith ".pdf":
           IF allow_pdf == false:
               continue
           # If you DO support PDFs, handle explicitly here; otherwise skip in prod.
           # (No PDF parsing in this stage by default.)
           continue

       # Accept only allowed types (prefix match: "text/html" etc.)
       is_allowed = false
       FOR each act in allowed_cts:
           IF ct startswith lower(act):
               is_allowed = true
               break
       IF is_allowed == false:
           continue

       # Decode HTML
       html = decode_html(resp.body_bytes, ct)
       IF html empty:
           continue

       # JS-heavy detection & policy
       IF is_js_heavy(html) == true:
           IF js_heavy_policy == "skip":
               continue
           IF js_heavy_policy == "fallback_reader":
               fb = reader_fallback(url)
               IF fb.status_code not in 200..299: continue
               html = fb.text
               IF html empty: continue
           # js_heavy_policy == "allow" => proceed with html as-is (may extract poorly)

       # Paywall detection & policy
       IF is_paywalled(html) == true:
           IF paywall_policy == "skip":
               continue
           IF paywall_policy == "extract_snippet":
               # Use search snippet only as a low-quality "window" chunk
               snippet = normalize_text(hit.snippet)
               IF snippet empty: continue
               h = hash_text(snippet)
               IF h in seen_chunk_hashes: continue
               seen_chunk_hashes.add(h)

               doc_id = make_doc_id(url)
               s = score_chunk(normalized_claim, snippet)

               candidate_items.append({
                 chunk_id: make_chunk_id(claim_id, doc_id, 0, 0),
                 chunk_kind: "window",
                 span: { p_start: 0, p_end: 0 },
                 text: snippet,
                 doc: {
                   doc_id: doc_id,
                   source: (hit.source if present else dom),
                   title: hit.title,
                   url: url,
                   published_at: hit.published_at?
                 },
                 score: s,
                 query_used: hit.query_used
               })
               continue
           # paywall_policy == "allow" => proceed to extraction (may be partial)

       doc_id = make_doc_id(url)

       # Paragraph extraction
       paras = extract_main_text_with_paragraphs(html)      # ordered
       paras = [ normalize_text(p) for p in paras if p not empty ]

       IF len(paras) >= 1:
           paras = merge_short_paragraphs(paras, PARA_MIN_CHARS)
           paras = split_long_paragraphs(paras, PARA_MAX_CHARS)   # optional

       # Fallback if paragraph extraction failed
       IF len(paras) < MIN_PARAS_FOR_SUCCESS:
           full_text = extract_main_text_plain(html)                # optional
           full_text = normalize_text(full_text)

           IF len(full_text) < MIN_TEXT_CHARS_FALLBACK:
               continue

           windows = window_chunk(full_text, FALLBACK_WINDOW_MAX, FALLBACK_WINDOW_OVERLAP)

           FOR each w in windows:
               w = normalize_text(w)
               IF w empty: continue
               h = hash_text(w)
               IF h in seen_chunk_hashes: continue
               seen_chunk_hashes.add(h)

               s = score_chunk(normalized_claim, w)

               candidate_items.append({
                 chunk_id: make_chunk_id(claim_id, doc_id, 0, 0),
                 chunk_kind: "window",
                 span: { p_start: 0, p_end: 0 },
                 text: w,
                 doc: {
                   doc_id: doc_id,
                   source: (hit.source if present else dom),
                   title: hit.title,
                   url: url,
                   published_at: hit.published_at?
                 },
                 score: s,
                 query_used: hit.query_used
               })

           continue   # next URL

       # Normal path: paragraph chunks with spans
       p = 0
       WHILE p < len(paras):
           chunk_text = paras[p]
           p_start = p
           p_end = p

           h = hash_text(chunk_text)
           IF h in seen_chunk_hashes:
               p = p + 1
               continue
           seen_chunk_hashes.add(h)

           s = score_chunk(normalized_claim, chunk_text)

           candidate_items.append({
             chunk_id: make_chunk_id(claim_id, doc_id, p_start, p_end),
             chunk_kind: "paragraph",
             span: { p_start: p_start, p_end: p_end },
             text: chunk_text,
             doc: {
               doc_id: doc_id,
               source: (hit.source if present else dom),
               title: hit.title,
               url: url,
               published_at: hit.published_at?
             },
             score: s,
             query_used: hit.query_used
           })

           p = p + 1

5) Form candidate pool of size ~ M
   # Not MMR. Just keep top-M by retrieval score.
   # Tie-breakers defined below.
   candidate_items.sort_by(
     score DESC,
     doc.doc_id ASC,
     chunk_id ASC
   )
   candidate_items = candidate_items[:M]

6) Compose rag_candidates
   rag_candidates = {
     pool_size_m: M,
     sources: ["web"],
     items: candidate_items
   }

7) Emit output (passthrough + rag_candidates)
   OUTPUT = INPUT plus "rag_candidates" field

RETURN OUTPUT

----------------------------------------
DETERMINISM / TIE-BREAKING (PROD)
----------------------------------------
Goal: avoid flaky outputs when sorting or taking argmax over floats.

Definitions:
  EPS = 1e-9  (treat |a-b| <= EPS as a tie)

Rules:
  1) Never rely on dict/map iteration order for ranking.
  2) Any argmax must define a deterministic tie-break order.
  3) Any sort must define secondary/tertiary keys (stable IDs).

Stage 5 (RAG retrieval):
  - When selecting top-M candidates by score:
      primary: score DESC
      tie-break 1: doc.doc_id ASC
      tie-break 2: chunk_id ASC
