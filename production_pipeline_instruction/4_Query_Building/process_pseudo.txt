# 4_Query_building/process_pseudo.txt
# Assumption: ai_model(system_instruction: string, claim_text: string) -> string exists.
# Goal: build 4 query variants = original + 3 meaning-preserving paraphrases (filtered).

INPUT:
  {
    request_id,
    claim_id,
    user_claim,
    normalized_claim,
    roberta: { label: { class_name }, confidence, embedding? },
    routing: { rag: { use_query_expansion, allow_web_search }, tier },
    meta: { received_at, version, warnings? }
  }

CONSTANTS (tunable):
  SYS_PARAPHRASE =
    "Without changing the meaning, paraphrase the following text. " +
    "Keep it short. Preserve names, numbers, dates, and negations."

  N_PARAPHRASES = 3
  MIN_COSINE = 0.85
  REQUIRE_NUMBERS_MATCH = true

  # If you don't have an embedding similarity function yet,
  # you can temporarily skip cosine filtering and only do heuristic checks.
  # But best practice is to filter.

HELPERS (assumed):
  extract_numbers(text) -> set<string>
  extract_entities_simple(text) -> set<string>        # optional heuristic (caps/NER-lite)
  embed(text) -> vector<float>                         # optional (bi-encoder)
  cosine(a, b) -> float

PROCESS:
1) Initialize base query
   original_q = normalized_claim
   queries = [{ q: original_q, variant: "original" }]

2) Decide whether to generate paraphrases
   paraphrase_enabled = routing.rag.use_query_expansion

   IF paraphrase_enabled == false:
       mode = "original_only"
       kept = 0
       GOTO step 6

3) Generate N paraphrases
   raw_paras = []
   FOR i in 1..N_PARAPHRASES:
       p = ai_model(SYS_PARAPHRASE, normalized_claim)
       p = strip(collapse_whitespace(p))
       IF p is empty: continue
       raw_paras.append(p)

4) Filter paraphrases (meaning-preservation guard)
   kept_paras = []
   nums0 = extract_numbers(normalized_claim)

   # optional semantic filter
   v0 = embed(normalized_claim)           # if embed() available, else set v0 = null

   FOR each p in raw_paras:
       # basic: de-dup + not identical
       IF p == normalized_claim: continue
       IF p already in kept_paras: continue

       # heuristic: numbers must match (if enabled)
       IF REQUIRE_NUMBERS_MATCH:
           IF extract_numbers(p) != nums0:
               continue

       # optional semantic similarity
       IF embed available:
           vp = embed(p)
           IF cosine(v0, vp) < MIN_COSINE:
               continue

       kept_paras.append(p)

       IF len(kept_paras) == N_PARAPHRASES:
           break

5) Add kept paraphrases to query list
   FOR each p in kept_paras:
       queries.append({ q: p, variant: "paraphrase" })

   mode = "original_plus_paraphrases"
   kept = len(kept_paras)

6) Build query_plan
   query_plan = {
     mode: mode,
     primary_query: original_q,
     queries: queries,                              # size 1..4
     paraphrase_meta: {
       enabled: paraphrase_enabled,
       n_requested: (paraphrase_enabled ? N_PARAPHRASES : 0),
       kept: kept,
       filter: {
         min_cosine: MIN_COSINE,
         require_numbers_match: REQUIRE_NUMBERS_MATCH
       }
     }
   }

7) Emit output (passthrough + query_plan)
   OUTPUT = INPUT plus "query_plan" field

RETURN OUTPUT
