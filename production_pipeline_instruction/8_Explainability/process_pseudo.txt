# 8_Explainability_SHAP/process_pseudo.txt
# Goal: compute token-level SHAP attributions for the RoBERTa claim-only prediction.
# Optimization: cache the SHAP masker + explainer per (model_name, revision, method, output_space, nsamples).

# Assumptions (implementation details ignored):
#   roberta_predict(texts: list<string>) -> array<float>[len(texts), 3]
#     returns probabilities (preferred) or logits for classes [false, mixed, true]
#   TOKENIZER is the exact tokenizer used by RoBERTa inference
#   SHAP is available:
#     shap.maskers.Text(tokenizer)
#     shap.Explainer(model_fn, masker, algorithm=METHOD)
#
# Cache:
#   EXPLAINER_CACHE: dict<string, {masker, explainer}>
#   cache_key = f"{model_name}:{revision}:{METHOD}:{OUTPUT_SPACE}:{NSAMPLES}"

INPUT:
  {
    request_id,
    claim_id,
    user_claim,
    normalized_claim,
    roberta: {
      label: { probs, class_id, class_name },
      model: { name, revision, ctx_tokens, T_used? },
      tokenization: { truncated, input_tokens }
    },
    routing,
    query_plan,
    rag_candidates,
    mmr_selected,
    evidence_topk,
    meta
  }

CONSTANTS (tunable):
  METHOD = "shap_partition"           # preferred for text
  OUTPUT_SPACE = "probability"        # align with roberta.label.probs
  NSAMPLES = 256
  TOPK_TOKENS = 8
  BACKGROUND_STRATEGY = "masker_default"

HELPERS (assumed):
  now_ms() -> int
  argsort_desc(values) -> list<int>
  argsort_asc(values)  -> list<int>
  clamp_int(x, lo, hi) -> int
  safe_token_spans(explanation, text) -> list<{start,end}> or null

GLOBAL (process-wide):
  EXPLAINER_CACHE = {}    # in-memory; can be LRU in real impl

PROCESS:
1) Determine explained class (use RoBERTa predicted class)
   explained_class_id = roberta.label.class_id
   explained_class_name = roberta.label.class_name

2) Build cache key
   model_name = roberta.model.name
   revision = roberta.model.revision
   cache_key = model_name + ":" + revision + ":" + METHOD + ":" + OUTPUT_SPACE + ":" + str(NSAMPLES)

3) Get or create cached masker + explainer
   IF cache_key in EXPLAINER_CACHE:
       masker = EXPLAINER_CACHE[cache_key].masker
       explainer = EXPLAINER_CACHE[cache_key].explainer
   ELSE:
       # define model function for SHAP (batch of masked texts)
       function f(text_list):
           out = roberta_predict(text_list)                 # shape [B, 3]
           # If roberta_predict returns logits and OUTPUT_SPACE == probability, apply softmax per row.
           # If it already returns probabilities, return as-is.
           return out_in_requested_space(out, OUTPUT_SPACE)

       masker = shap.maskers.Text(TOKENIZER)
       explainer = shap.Explainer(f, masker, algorithm=METHOD)

       EXPLAINER_CACHE[cache_key] = { masker: masker, explainer: explainer }

4) Run SHAP on the single instance
   start_ms = now_ms()
   explanation = explainer([normalized_claim], nsamples=NSAMPLES)
   time_ms = now_ms() - start_ms

5) Extract token data + SHAP values for the explained class
   # Typical shapes:
   #   explanation.data[0]          -> list of token strings/segments
   #   explanation.values[0]        -> [num_tokens, num_classes]
   token_strings = explanation.data[0]
   shap_vals_all = explanation.values[0]                     # matrix
   shap_values = shap_vals_all[:, explained_class_id]        # vector length = num_tokens

6) Determine base_value and output_value
   # base_values often: [num_classes]
   base_value = explanation.base_values[0][explained_class_id]

   # output_value may not always be present in all SHAP objects; fall back safely
   IF explanation has "output_values":
       output_value = explanation.output_values[0][explained_class_id]
   ELSE:
       output_value = roberta.label.probs[explained_class_id]    # safe fallback

7) Optional: compute token char spans (best-effort)
   spans = safe_token_spans(explanation, normalized_claim)   # returns list of {start,end} or null

8) Build token records
   tokens_out = []
   FOR i in 0..len(token_strings)-1:
       rec = { i: i, token: token_strings[i], shap_value: shap_values[i] }
       IF spans is not null AND i < len(spans):
           rec.char_span = { start: spans[i].start, end: spans[i].end }
       tokens_out.append(rec)

9) Compute top tokens (indices)
   k = clamp_int(TOPK_TOKENS, 1, len(shap_values))
   idx_pos = argsort_desc(shap_values)[0:k]
   idx_neg = argsort_asc(shap_values)[0:k]

10) Compose shap_explainability block
   shap_explainability = {
     method: METHOD,
     explained_class: { class_id: explained_class_id, class_name: explained_class_name },
     base_value: base_value,
     output_value: output_value,
     output_space: OUTPUT_SPACE,
     tokens: tokens_out,
     top_tokens: { k: k, positive: idx_pos, negative: idx_neg },
     meta: { background: BACKGROUND_STRATEGY, nsamples: NSAMPLES, time_ms: time_ms }
   }

11) Emit output (passthrough + shap_explainability)
   OUTPUT = INPUT plus "shap_explainability" field

RETURN OUTPUT

----------------------------------------
DETERMINISM / TIE-BREAKING (PROD)
----------------------------------------
Goal: avoid flaky outputs when sorting or taking argmax over floats.

Definitions:
  EPS = 1e-9  (treat |a-b| <= EPS as a tie)

Rules:
  1) Never rely on dict/map iteration order for ranking.
  2) Any argmax must define a deterministic tie-break order.
  3) Any sort must define secondary/tertiary keys (stable IDs).

Stage 8 (SHAP explainability):
  - When ranking tokens by importance:
      primary: abs(shap_value) DESC
      tie-break 1: token_index ASC (earlier token wins)
