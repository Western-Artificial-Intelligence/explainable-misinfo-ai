# 7_Light_relevance_gate/process_pseudo.txt
# Goal: apply a lightweight relevance filter on MMR-selected chunks and keep top-K.
# This is NOT a verifier. It only removes obviously-irrelevant chunks.

# Assumptions (implementation details ignored):
#   embed(text: string, model: string) -> vector<float>
#   cosine(a,b) -> float
#   OPTIONAL: cross_encoder_score(claim, chunk) -> float in [0,1]
#
#   embedding_cache_get(key: string) -> optional<vector<float>>
#   embedding_cache_set(key: string, vec: vector<float>, ttl_s: int) -> void
#
#   index_candidates_by_chunk_id(rag_candidates.items) -> map<chunk_id, item>

MIN_RELEVANCE METRIC (definition):
  - Let ec = embed(normalized_claim) and ek = embed(chunk_text) using the SAME embed_model.
  - L2-normalize both vectors, then relevance = cosine(ec, ek) in [0,1] (after normalization).
  - Keep chunk if relevance >= T where T = routing.rag.min_relevance.


INPUT:
  {
    request_id,
    claim_id,
    user_claim,
    normalized_claim,
    roberta,
    routing: { rag: { final_top_k (K), min_relevance (T) } },
    rag_candidates: { items: [...] },
    mmr_selected: { items: [ {chunk_id, ...} ] },
    meta: {
      # optional efficiency knobs (recommended)
      embed_model?,
      embedding_cache_enabled?,
      embedding_cache_namespace?,
      embedding_cache_ttl_s?,
      ...
    }
  }

CONSTANTS (tunable defaults if meta.* not provided):
  GATE_METHOD = "cosine_threshold"      # or "cross_encoder_threshold"
  DEFAULT_EMBED_MODEL = "sentence-transformers/all-MiniLM-L6-v2"
  DEFAULT_EMBED_CACHE_ENABLED = true
  DEFAULT_EMBED_CACHE_NAMESPACE = "emb"
  DEFAULT_EMBED_CACHE_TTL_S = 0
  EPS = 1e-9

HELPERS:
  cache_key(embed_model, chunk_id):
    ns = (meta.embedding_cache_namespace OR DEFAULT_EMBED_CACHE_NAMESPACE)
    return ns + ":" + embed_model + ":" + chunk_id

PROCESS:
1) Index full candidate items by chunk_id (to recover text + doc metadata)
   by_id = index_candidates_by_chunk_id(rag_candidates.items)

2) Resolve embedding cache configuration
   embed_model = meta.embed_model OR DEFAULT_EMBED_MODEL
   cache_enabled = (meta.embedding_cache_enabled if present else DEFAULT_EMBED_CACHE_ENABLED)
   cache_ttl_s = meta.embedding_cache_ttl_s OR DEFAULT_EMBED_CACHE_TTL_S

3) Load / compute claim embedding (reuse Stage 6 cache if available)
   q_key = cache_key(embed_model, "claim:" + claim_id)
   IF cache_enabled:
       q_vec = embedding_cache_get(q_key)
   IF q_vec missing:
       q_vec = embed(normalized_claim, embed_model)
       IF cache_enabled:
           embedding_cache_set(q_key, q_vec, cache_ttl_s)

4) Score each MMR-selected chunk for relevance
   scored = []

   FOR each sel in mmr_selected.items:
       cid = sel.chunk_id
       base = by_id.get(cid)
       IF base missing:
           continue

       text = base.text

       IF GATE_METHOD == "cosine_threshold":
           k = cache_key(embed_model, cid)
           IF cache_enabled:
               v = embedding_cache_get(k)
           IF v missing:
               # In best practice, this should almost always hit because Stage 6 cached it.
               v = embed(text, embed_model)
               IF cache_enabled:
                   embedding_cache_set(k, v, cache_ttl_s)

           rel = cosine(v, q_vec)      # [-1,1] depending on embedder; threshold should match embedder behavior

       ELSE IF GATE_METHOD == "cross_encoder_threshold":
           rel = cross_encoder_score(normalized_claim, text)   # [0,1]

       # Keep only those above threshold T (or keep all and sort if desired)
       IF rel + EPS < routing.rag.min_relevance:
           continue

       scored.append({
         "chunk_id": cid,
         "relevance_score": rel,
         "retrieval_score": base.score,
         "doc": base.doc,
         "text": text
       })

5) Select top-K with deterministic tie-breakers
   K = routing.rag.final_top_k

   # Sort:
   #   primary: relevance_score DESC
   #   tie-break 1: original retrieval_score DESC
   #   tie-break 2: chunk_id ASC
   scored.sort_by(
     relevance_score DESC,
     retrieval_score DESC,
     chunk_id ASC
   )

   kept = scored[:K]

6) Build evidence_topk output (ranked)
   evidence_topk = {
     top_k: K,
     gate_method: GATE_METHOD,
     embed_model: embed_model,
     items: [
       {
         rank: i+1,
         chunk_id: item.chunk_id,
         doc_id: item.doc.doc_id,
         text: item.text,
         relevance_score: item.relevance_score,
         score: item.retrieval_score,
         doc: item.doc
       } for i,item in enumerate(kept)
     ]
   }

7) Output (passthrough + evidence_topk)
   OUTPUT = INPUT plus "evidence_topk" field
   RETURN OUTPUT
